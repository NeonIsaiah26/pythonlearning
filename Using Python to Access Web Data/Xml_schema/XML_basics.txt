# Start Tag : <person>, <name>, <phone>
# End tag : </person>, </name>, </phone>
# Text Content : Chuck, +1 734 303 4456
# Attribute : type="intl", hide="yes"
# Self closing tag : <email {attribute}/> 

# White space doesn't matter 

<person>
    <name>Chuck</name>
    <phone type = "intl">
        +1 734 303 4456
    </phone>
    <email hide="yes"/>
</person>
